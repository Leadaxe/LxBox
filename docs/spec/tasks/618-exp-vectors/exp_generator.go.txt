// expChromePQClientHello builds the Chrome 133 QUIC ClientHello via uTLS, keeping
// the post-quantum key share (unlike buildBrowserClientHello, which strips it).
func expChromePQClientHello(t *testing.T, sni string) []byte {
	t.Helper()
	spec, err := utls.UTLSIdToSpec(utls.HelloChrome_133)
	require.NoError(t, err)
	spec.TLSVersMin = utls.VersionTLS13
	spec.TLSVersMax = utls.VersionTLS13
	for _, ext := range spec.Extensions {
		switch e := ext.(type) {
		case *utls.ALPNExtension:
			e.AlpnProtocols = []string{"h3"}
		case *utls.ApplicationSettingsExtensionNew:
			e.SupportedProtocols = []string{"h3"}
		case *utls.SupportedVersionsExtension:
			e.Versions = []uint16{utls.VersionTLS13}
		}
	}
	cfg := &utls.Config{ServerName: sni, MinVersion: utls.VersionTLS13}
	q := utls.UQUICClient(&utls.QUICConfig{TLSConfig: cfg}, utls.HelloCustom)
	defer q.Close()
	require.NoError(t, q.ApplyPreset(&spec))
	q.SetTransportParameters(quicDecoyTransportParams())
	require.NoError(t, q.Start(context.Background()))
	var hello []byte
	for {
		ev := q.NextEvent()
		if ev.Kind == utls.QUICNoEvent {
			break
		}
		if ev.Kind == utls.QUICWriteData && ev.Level == utls.QUICEncryptionLevelInitial {
			hello = append(hello, ev.Data...)
		}
	}
	require.NotEmpty(t, hello)
	require.Equal(t, byte(0x01), hello[0])
	return hello
}

// expBuildInitial assembles ONE Initial: header(dcid, pn) ‖ CRYPTO[off] ‖ PADDING
// to totalLen. Returns the packet and how many ClientHello bytes it consumed.
func expBuildInitial(t *testing.T, dcid []byte, pn uint64, ch []byte, off int, totalLen int) ([]byte, int) {
	t.Helper()
	const headerFixed = 1 + 4 + 1 + 1 + 1
	const lenFieldVarintWidth = 2
	headerLen := headerFixed + quicDCIDLen + lenFieldVarintWidth + quicPacketNumberLen
	payloadLen := totalLen - headerLen - quicAEADTagLen
	lengthField := quicPacketNumberLen + payloadLen + quicAEADTagLen
	require.True(t, lengthField >= 1<<6 && lengthField < 1<<14)

	rest := ch[off:]
	// CRYPTO frame overhead: type(1) + varint(off) + varint(len). len varint is
	// 2 bytes for anything ≥64, which every slice here is.
	maxData := payloadLen - (1 + varintLen(uint64(off)) + 2)
	n := len(rest)
	if n > maxData {
		n = maxData
	}
	frag := cryptoFragment{offset: uint64(off), data: rest[:n]}
	payload := appendCryptoFrame(make([]byte, 0, payloadLen), frag)
	for len(payload) < payloadLen {
		payload = append(payload, 0x00)
	}
	require.Equal(t, payloadLen, len(payload))

	header := make([]byte, 0, headerLen)
	header = append(header, 0xC0|byte(quicPacketNumberLen-1))
	header = binary.BigEndian.AppendUint32(header, quicVersion1)
	header = append(header, byte(quicDCIDLen))
	header = append(header, dcid...)
	header = append(header, 0x00, 0x00)
	header = appendQUICVarint(header, uint64(lengthField))
	pnOffset := len(header)
	for i := 0; i < quicPacketNumberLen; i++ {
		header = append(header, byte(pn>>(8*(quicPacketNumberLen-1-i))))
	}
	key, iv, hp := deriveInitialKeys(dcid)
	pkt, err := encryptInitial(header, payload, key, iv, hp, pnOffset, pn)
	require.NoError(t, err)
	require.Equal(t, totalLen, len(pkt))
	return pkt, n
}

// expDecrypt reverse-parses an Initial without requiring the CRYPTO data to
// start at offset 0 (decryptInitial does). Returns pn, frame walk, CRYPTO frags.
func expDecrypt(t *testing.T, packet []byte) (uint64, string, []cryptoFragment) {
	t.Helper()
	r := bytes.NewReader(packet)
	first, _ := r.ReadByte()
	require.Equal(t, byte(0xC0), first&0xF0)
	var version uint32
	require.NoError(t, binary.Read(r, binary.BigEndian, &version))
	require.Equal(t, quicVersion1, version)
	dcidLen, _ := r.ReadByte()
	dcid := make([]byte, dcidLen)
	_, err := io.ReadFull(r, dcid)
	require.NoError(t, err)
	scidLen, _ := r.ReadByte()
	require.Equal(t, byte(0), scidLen)
	tokenLen, _ := r.ReadByte()
	require.Equal(t, byte(0), tokenLen)
	lengthField := readVarint(t, r)
	hdrLen := len(packet) - r.Len()

	key, iv, hp := deriveInitialKeys(dcid)
	block, err := aes.NewCipher(hp)
	require.NoError(t, err)
	mask := make([]byte, aes.BlockSize)
	block.Encrypt(mask, packet[hdrLen+4:hdrLen+4+aes.BlockSize])
	un := append([]byte(nil), packet...)
	un[0] ^= mask[0] & 0x0f
	pnLen := int(un[0]&0x03) + 1
	var pn uint64
	for i := 0; i < pnLen; i++ {
		un[hdrLen+i] ^= mask[1+i]
		pn = pn<<8 | uint64(un[hdrLen+i])
	}
	aad := un[:hdrLen+pnLen]
	ct := un[hdrLen+pnLen : hdrLen+int(lengthField)]
	cipher := quicAEADAESGCMTLS13(key, iv)
	nonce := make([]byte, cipher.NonceSize())
	binary.BigEndian.PutUint64(nonce[cipher.NonceSize()-8:], pn)
	dec, err := cipher.Open(nil, nonce, ct, aad)
	require.NoError(t, err, "AEAD open")

	var walk []string
	var frags []cryptoFragment
	fr := bytes.NewReader(dec)
	pad := 0
	flushPad := func() {
		if pad > 0 {
			walk = append(walk, fmt.Sprintf("PAD×%d", pad))
			pad = 0
		}
	}
	for {
		ft, err := fr.ReadByte()
		if err == io.EOF {
			break
		}
		switch ft {
		case 0x00:
			pad++
		case 0x01:
			flushPad()
			walk = append(walk, "PING")
		case 0x06:
			flushPad()
			off := readVarint(t, fr)
			ln := readVarint(t, fr)
			idx := len(dec) - fr.Len()
			frags = append(frags, cryptoFragment{offset: off, data: append([]byte(nil), dec[idx:idx+int(ln)]...)})
			_, _ = fr.Seek(int64(ln), io.SeekCurrent)
			walk = append(walk, fmt.Sprintf("CRYPTO[off=%d len=%d]", off, ln))
		default:
			t.Fatalf("unexpected frame 0x%02x", ft)
		}
	}
	flushPad()
	return pn, strings.Join(walk, " "), frags
}

// expHasMLKEMKeyShare scans the ClientHello's key_share extension for group 4588.
func expHasMLKEMKeyShare(t *testing.T, ch []byte) bool {
	t.Helper()
	r := bytes.NewReader(ch[4:])
	skip(t, r, 2+32)
	sid := readU8(t, r)
	skip(t, r, int(sid))
	cs := readU16(t, r)
	skip(t, r, int(cs))
	comp := readU8(t, r)
	skip(t, r, int(comp))
	extTotal := readU16(t, r)
	end := r.Len() - int(extTotal)
	for r.Len() > end {
		et := readU16(t, r)
		el := readU16(t, r)
		body := make([]byte, el)
		_, err := io.ReadFull(r, body)
		require.NoError(t, err)
		if et != 0x0033 {
			continue
		}
		br := bytes.NewReader(body)
		listLen := readU16(t, br)
		lend := br.Len() - int(listLen)
		for br.Len() > lend {
			g := readU16(t, br)
			l := readU16(t, br)
			skip(t, br, int(l))
			if g == uint16(utls.X25519MLKEM768) {
				return true
			}
		}
	}
	return false
}

func TestExpArmM(t *testing.T) {
	out := os.Getenv("EXP_OUT")
	if out == "" {
		t.Skip("EXP_OUT not set")
	}
	require.NoError(t, os.MkdirAll(out, 0o755))
	var walkLog strings.Builder

	for _, sni := range []string{"apteka.ru", "wartune.mail.ru"} {
		ch := expChromePQClientHello(t, sni)
		require.True(t, expHasMLKEMKeyShare(t, ch), "ClientHello must carry X25519MLKEM768 key share")

		dcid := make([]byte, quicDCIDLen)
		_, err := rand.Read(dcid)
		require.NoError(t, err)

		p0, n0 := expBuildInitial(t, dcid, 0, ch, 0, quicInitialTotalLen)
		require.Less(t, n0, len(ch), "arm M needs a ClientHello that does NOT fit one Initial")
		p1, n1 := expBuildInitial(t, dcid, 1, ch, n0, quicInitialTotalLen)
		require.Equal(t, len(ch), n0+n1, "two Initials must carry the whole ClientHello")

		// Reverse-check both packets and the cross-packet reassembly.
		pn0, w0, f0 := expDecrypt(t, p0)
		pn1, w1, f1 := expDecrypt(t, p1)
		require.Equal(t, uint64(0), pn0)
		require.Equal(t, uint64(1), pn1)
		require.Len(t, f0, 1)
		require.Len(t, f1, 1)
		require.Equal(t, uint64(0), f0[0].offset)
		require.Equal(t, uint64(n0), f1[0].offset)
		re := reassemble(t, append(f0, f1...))
		require.Equal(t, ch, re)
		require.Equal(t, sni, extractSNI(t, re))

		tag := strings.ReplaceAll(sni, ".", "_")
		require.NoError(t, os.WriteFile(filepath.Join(out, "i1_M_"+tag+".cps"), []byte("<b 0x"+hex.EncodeToString(p0)+">"), 0o644))
		require.NoError(t, os.WriteFile(filepath.Join(out, "i2_M_"+tag+".cps"), []byte("<b 0x"+hex.EncodeToString(p1)+">"), 0o644))
		fmt.Fprintf(&walkLog, "M %s: ch=%d (Chrome 133 uTLS, X25519MLKEM768 kept, ALPN h3) dcid=%x\n", sni, len(ch), dcid)
		fmt.Fprintf(&walkLog, "  i1: pkt=%d pn=0 frames: %s\n", len(p0), w0)
		fmt.Fprintf(&walkLog, "  i2: pkt=%d pn=1 frames: %s\n", len(p1), w1)
	}
	require.NoError(t, os.WriteFile(filepath.Join(out, "i1i2_M_walk.txt"), []byte(walkLog.String()), 0o644))
	t.Log("\n" + walkLog.String())
}

// TestExpArmMInspect decrypts i1/i2 CPS blobs from EXP_IN (space-separated list
// of "label:i1path:i2path") and prints the ClientHello extension map against
// the packet-1 boundary (which bytes travel in which Initial).
func TestExpArmMInspect(t *testing.T) {
	in := os.Getenv("EXP_IN")
	if in == "" {
		t.Skip("EXP_IN not set")
	}
	for _, item := range strings.Fields(in) {
		parts := strings.Split(item, ":")
		require.Len(t, parts, 3)
		load := func(p string) []byte {
			raw, err := os.ReadFile(p)
			require.NoError(t, err)
			s := strings.TrimSpace(string(raw))
			s = strings.TrimPrefix(s, "<b 0x")
			s = strings.TrimSuffix(s, ">")
			b, err := hex.DecodeString(s)
			require.NoError(t, err)
			return b
		}
		_, _, f0 := expDecrypt(t, load(parts[1]))
		_, _, f1 := expDecrypt(t, load(parts[2]))
		ch := reassemble(t, append(f0, f1...))
		boundary := len(f0[0].data)
		// Walk extensions with absolute offsets inside ch.
		r := bytes.NewReader(ch[4:])
		skip(t, r, 2+32)
		sid := readU8(t, r)
		skip(t, r, int(sid))
		cs := readU16(t, r)
		skip(t, r, int(cs))
		comp := readU8(t, r)
		skip(t, r, int(comp))
		_ = readU16(t, r)
		var sb strings.Builder
		fmt.Fprintf(&sb, "%s ch=%d boundary=%d sni=%s ext:", parts[0], len(ch), boundary, extractSNI(t, ch))
		for r.Len() > 0 {
			start := len(ch) - r.Len()
			et := readU16(t, r)
			el := readU16(t, r)
			skip(t, r, int(el))
			end := len(ch) - r.Len()
			where := "p1"
			if start >= boundary {
				where = "p2"
			} else if end > boundary {
				where = "SPLIT"
			}
			name := fmt.Sprintf("0x%04x", et)
			switch et {
			case 0x0000:
				name = "SNI"
			case 0x0033:
				name = "key_share"
			case 0x0039:
				name = "quic_tp"
			case 0x0010:
				name = "alpn"
			case 0xfe0d:
				name = "ech"
			case 0x002b:
				name = "versions"
			case 0x000a:
				name = "groups"
			}
			if et&0x0f0f == 0x0a0a {
				name = "GREASE"
			}
			fmt.Fprintf(&sb, " %s[%d-%d %s]", name, start, end, where)
		}
		t.Log(sb.String())
	}
}

// TestExpArmMRepack re-packs an existing i1/i2 pair (EXP_REPACK="label:i1:i2:outdir")
// with a fresh DCID but the IDENTICAL ClientHello bytes, writing i1_<label>.cps /
// i2_<label>.cps into outdir. Isolates "DCID matters" from "ClientHello matters".
func TestExpArmMRepack(t *testing.T) {
	in := os.Getenv("EXP_REPACK")
	if in == "" {
		t.Skip("EXP_REPACK not set")
	}
	for _, item := range strings.Fields(in) {
		parts := strings.Split(item, ":")
		require.Len(t, parts, 4)
		load := func(p string) []byte {
			raw, err := os.ReadFile(p)
			require.NoError(t, err)
			s := strings.TrimSuffix(strings.TrimPrefix(strings.TrimSpace(string(raw)), "<b 0x"), ">")
			b, err := hex.DecodeString(s)
			require.NoError(t, err)
			return b
		}
		_, _, f0 := expDecrypt(t, load(parts[1]))
		_, _, f1 := expDecrypt(t, load(parts[2]))
		ch := reassemble(t, append(f0, f1...))
		dcid := make([]byte, quicDCIDLen)
		_, err := rand.Read(dcid)
		require.NoError(t, err)
		p0, n0 := expBuildInitial(t, dcid, 0, ch, 0, quicInitialTotalLen)
		p1, n1 := expBuildInitial(t, dcid, 1, ch, n0, quicInitialTotalLen)
		require.Equal(t, len(ch), n0+n1)
		require.NoError(t, os.MkdirAll(parts[3], 0o755))
		require.NoError(t, os.WriteFile(filepath.Join(parts[3], "i1_"+parts[0]+".cps"), []byte("<b 0x"+hex.EncodeToString(p0)+">"), 0o644))
		require.NoError(t, os.WriteFile(filepath.Join(parts[3], "i2_"+parts[0]+".cps"), []byte("<b 0x"+hex.EncodeToString(p1)+">"), 0o644))
		t.Logf("%s repacked ch=%d new dcid=%x", parts[0], len(ch), dcid)
	}
}

// expChromeNoPQClientHello: Chrome 133 via uTLS with the PQ key share stripped
// (fits one Initial), i.e. the current ib=chrome shape but on the 133 fingerprint.
func expChromeNoPQClientHello(t *testing.T, sni string) []byte {
	t.Helper()
	spec, err := utls.UTLSIdToSpec(utls.HelloChrome_133)
	require.NoError(t, err)
	spec.TLSVersMin = utls.VersionTLS13
	spec.TLSVersMax = utls.VersionTLS13
	for _, ext := range spec.Extensions {
		switch e := ext.(type) {
		case *utls.ALPNExtension:
			e.AlpnProtocols = []string{"h3"}
		case *utls.ApplicationSettingsExtensionNew:
			e.SupportedProtocols = []string{"h3"}
		case *utls.SupportedVersionsExtension:
			e.Versions = []uint16{utls.VersionTLS13}
		case *utls.SupportedCurvesExtension:
			var cs []utls.CurveID
			for _, c := range e.Curves {
				if notPQCurve(c) {
					cs = append(cs, c)
				}
			}
			e.Curves = cs
		case *utls.KeyShareExtension:
			var ks []utls.KeyShare
			for _, k := range e.KeyShares {
				if notPQCurve(k.Group) {
					ks = append(ks, k)
				}
			}
			e.KeyShares = ks
		}
	}
	cfg := &utls.Config{ServerName: sni, MinVersion: utls.VersionTLS13}
	q := utls.UQUICClient(&utls.QUICConfig{TLSConfig: cfg}, utls.HelloCustom)
	defer q.Close()
	require.NoError(t, q.ApplyPreset(&spec))
	q.SetTransportParameters(quicDecoyTransportParams())
	require.NoError(t, q.Start(context.Background()))
	var hello []byte
	for {
		ev := q.NextEvent()
		if ev.Kind == utls.QUICNoEvent {
			break
		}
		if ev.Kind == utls.QUICWriteData && ev.Level == utls.QUICEncryptionLevelInitial {
			hello = append(hello, ev.Data...)
		}
	}
	require.NotEmpty(t, hello)
	return hello
}

// expBuildInitialN is expBuildInitial with an explicit byte budget for the CRYPTO
// data (to force a split of a ClientHello that would otherwise fit).
func expBuildInitialN(t *testing.T, dcid []byte, pn uint64, ch []byte, off, take int, totalLen int) []byte {
	t.Helper()
	sub := ch[:off+take]
	pkt, n := expBuildInitial(t, dcid, pn, sub, off, totalLen)
	require.Equal(t, take, n)
	return pkt
}

func expWriteCPS(t *testing.T, dir, name string, pkt []byte) {
	t.Helper()
	require.NoError(t, os.WriteFile(filepath.Join(dir, name+".cps"), []byte("<b 0x"+hex.EncodeToString(pkt)+">"), 0o644))
}

// TestExpMatrix: for each domain in EXP_DOMAINS (space-separated) write into
// EXP_OUT: G2 (generic 294B CH split over two in-order Initials), K1 (Chrome 133
// without PQ, one Initial, CRYPTO off=0 + PADDING), M (Chrome 133 with PQ, two
// Initials). File names: <arm>_<domaintag>_i1.cps / _i2.cps.
func TestExpMatrix(t *testing.T) {
	out := os.Getenv("EXP_OUT")
	domains := strings.Fields(os.Getenv("EXP_DOMAINS"))
	if out == "" || len(domains) == 0 {
		t.Skip("EXP_OUT / EXP_DOMAINS not set")
	}
	require.NoError(t, os.MkdirAll(out, 0o755))
	var log strings.Builder
	for _, sni := range domains {
		tag := strings.ReplaceAll(sni, ".", "_")
		newDCID := func() []byte {
			d := make([]byte, quicDCIDLen)
			_, err := rand.Read(d)
			require.NoError(t, err)
			return d
		}
		// G2: generic CH (same builder as the product default), split in two.
		var tlsRandom [32]byte
		_, err := rand.Read(tlsRandom[:])
		require.NoError(t, err)
		ecKey, err := ecdh.X25519().GenerateKey(rand.Reader)
		require.NoError(t, err)
		gch, err := buildGenericClientHello(sni, tlsRandom, ecKey.PublicKey().Bytes())
		require.NoError(t, err)
		d := newDCID()
		cut := len(gch) / 2
		g1 := expBuildInitialN(t, d, 0, gch, 0, cut, quicInitialTotalLen)
		g2, n2 := expBuildInitial(t, d, 1, gch, cut, quicInitialTotalLen)
		require.Equal(t, len(gch)-cut, n2)
		_, w1, f1 := expDecrypt(t, g1)
		_, w2, f2 := expDecrypt(t, g2)
		require.Equal(t, gch, reassemble(t, append(f1, f2...)))
		expWriteCPS(t, out, "G2_"+tag+"_i1", g1)
		expWriteCPS(t, out, "G2_"+tag+"_i2", g2)
		fmt.Fprintf(&log, "G2 %s: ch=%d dcid=%x\n  i1: %s\n  i2: %s\n", sni, len(gch), d, w1, w2)

		// K1: Chrome 133 without PQ, one Initial.
		kch := expChromeNoPQClientHello(t, sni)
		d = newDCID()
		k1, nk := expBuildInitial(t, d, 0, kch, 0, quicInitialTotalLen)
		require.Equal(t, len(kch), nk, "K1 must fit one Initial")
		_, wk, fk := expDecrypt(t, k1)
		require.Equal(t, sni, extractSNI(t, reassemble(t, fk)))
		expWriteCPS(t, out, "K1_"+tag+"_i1", k1)
		fmt.Fprintf(&log, "K1 %s: ch=%d dcid=%x\n  i1: %s\n", sni, len(kch), d, wk)

		// M: Chrome 133 with PQ, two Initials.
		mch := expChromePQClientHello(t, sni)
		d = newDCID()
		m1, n0 := expBuildInitial(t, d, 0, mch, 0, quicInitialTotalLen)
		m2, n1 := expBuildInitial(t, d, 1, mch, n0, quicInitialTotalLen)
		require.Equal(t, len(mch), n0+n1)
		_, wm1, _ := expDecrypt(t, m1)
		_, wm2, _ := expDecrypt(t, m2)
		expWriteCPS(t, out, "M_"+tag+"_i1", m1)
		expWriteCPS(t, out, "M_"+tag+"_i2", m2)
		fmt.Fprintf(&log, "M %s: ch=%d dcid=%x\n  i1: %s\n  i2: %s\n", sni, len(mch), d, wm1, wm2)
	}
	require.NoError(t, os.WriteFile(filepath.Join(out, "matrix_walk.txt"), []byte(log.String()), 0o644))
	t.Log("\n" + log.String())
}

			s := strings.TrimSuffix(strings.TrimPrefix(strings.TrimSpace(string(raw)), "<b 0x"), ">")
			b, err := hex.DecodeString(s)
			require.NoError(t, err)
			return b
		}
		_, _, f1 := expDecrypt(t, load(parts[2]))
		require.Len(t, f1, 1)
		off := int(f1[0].offset)
		full := make([]byte, off+len(f1[0].data))
		copy(full[off:], f1[0].data)
		dcid := make([]byte, quicDCIDLen)
		_, err := rand.Read(dcid)
		require.NoError(t, err)
		p1, n := expBuildInitial(t, dcid, 1, full, off, quicInitialTotalLen)
		require.Equal(t, len(f1[0].data), n)
		require.NoError(t, os.MkdirAll(parts[3], 0o755))
		expWriteCPS(t, parts[3], "i2_"+parts[0]+"_newdcid", p1)
		t.Logf("%s: i2 re-encrypted under dcid=%x (off=%d len=%d)", parts[0], dcid, off, n)
	}
}

// TestExpArmS: Chrome 133 + MLKEM, extension order fixed (no shuffle) so SNI sits
// early; the FIRST Initial is cut INSIDE the SNI host name at a label boundary
// (after "<first label>."), the rest follows in two more in-order Initials.
// EXP_OUT, EXP_DOMAINS as in TestExpMatrix. Files: S_<tag>_i1/_i2/_i3.cps.
func TestExpArmS(t *testing.T) {
	out := os.Getenv("EXP_OUT")
	domains := strings.Fields(os.Getenv("EXP_DOMAINS"))

		t.Skip("EXP_OUT / EXP_DOMAINS not set")
	}

		spec, err := utls.UTLSIdToSpec(utls.HelloChrome_133)
		require.NoError(t, err)
		spec.TLSVersMin, spec.TLSVersMax = utls.VersionTLS13, utls.VersionTLS13
		// Fixed order: GREASE, SNI, then the rest as listed (shuffle undone by
		// moving SNI right after the leading GREASE).
		var sniExt utls.TLSExtension
		var rest []utls.TLSExtension

			switch e := ext.(type) {

	}

	t.Log("\n" + log.String())

// TestExpArmR cuts the ClientHello into in-order CRYPTO frames whose boundaries
// fall on the dots of the SNI host name, then packs those frames in order into
// as many Initials as needed (greedy fill). EXP_OUT / EXP_DOMAINS as usual.
func TestExpArmR(t *testing.T) {
	out := os.Getenv("EXP_OUT")
	domains := strings.Fields(os.Getenv("EXP_DOMAINS"))
	if out == "" || len(domains) == 0 {
		t.Skip("EXP_OUT / EXP_DOMAINS not set")
	}
	require.NoError(t, os.MkdirAll(out, 0o755))
	var log strings.Builder
	const budget = 1211 // CRYPTO payload room in one 1250-byte Initial (approx)
	for _, sni := range domains {
		tag := strings.ReplaceAll(sni, ".", "_")
		ch, host := expChrome133SNIEarly(t, sni)
		require.True(t, expHasMLKEMKeyShare(t, ch))

		// Fragment boundaries: start, after each dot inside the host name, end.
		bounds := []int{0}
		for i, c := range sni {
			if c == '.' {
				bounds = append(bounds, host+i+1)
			}
		}
		bounds = append(bounds, len(ch))
		var frags []cryptoFragment
		for i := 0; i+1 < len(bounds); i++ {
			frags = append(frags, cryptoFragment{offset: uint64(bounds[i]), data: ch[bounds[i]:bounds[i+1]]})
		}

		// Greedy-pack consecutive frames into Initials, splitting a frame only
		// when it alone exceeds the budget.
		dcid := make([]byte, quicDCIDLen)
		_, err := rand.Read(dcid)
		require.NoError(t, err)
		var pkts [][]byte
		var cur []cryptoFragment
		curLen := 0
		flush := func() {
			if len(cur) == 0 {
				return
			}
			pkts = append(pkts, expBuildInitialFrames(t, dcid, uint64(len(pkts)), cur, quicInitialTotalLen))
			cur, curLen = nil, 0
		}
		for _, f := range frags {
			data := f.data
			off := f.offset
			for len(data) > 0 {
				cost := 1 + varintLen(off) + varintLen(uint64(len(data))) + len(data)
				if curLen+cost <= budget {
					cur = append(cur, cryptoFragment{offset: off, data: data})
					curLen += cost
					break
				}
				room := budget - curLen - (1 + varintLen(off) + 2)
				if room <= 0 {
					flush()
					continue
				}
				cur = append(cur, cryptoFragment{offset: off, data: data[:room]})
				flush()
				off += uint64(room)
				data = data[room:]
			}
		}
		flush()

		// Reverse-check: decrypt every Initial, reassemble, verify SNI.
		var all []cryptoFragment
		var walk []string
		for i, p := range pkts {
			_, w, fs := expDecrypt(t, p)
			all = append(all, fs...)
			walk = append(walk, fmt.Sprintf("i%d: %s", i+1, w))
			expWriteCPS(t, out, fmt.Sprintf("R_%s_i%d", tag, i+1), p)
		}
		require.Equal(t, ch, reassemble(t, all))
		require.Equal(t, sni, extractSNI(t, reassemble(t, all)))
		fmt.Fprintf(&log, "R %s: ch=%d sni@%d frames=%d packets=%d dcid=%x\n  %s\n", sni, len(ch), host, len(frags), len(pkts), dcid, strings.Join(walk, "\n  "))
	}
	require.NoError(t, os.WriteFile(filepath.Join(out, "R_walk.txt"), []byte(log.String()), 0o644))
	t.Log("\n" + log.String())
}

// expBuildInitialFrames packs in-order CRYPTO fragments into one Initial, padding
// to totalLen. Caller guarantees they fit the payload budget.
func expBuildInitialFrames(t *testing.T, dcid []byte, pn uint64, frags []cryptoFragment, totalLen int) []byte {
	t.Helper()
	const headerFixed = 1 + 4 + 1 + 1 + 1
	headerLen := headerFixed + quicDCIDLen + 2 + quicPacketNumberLen
	payloadLen := totalLen - headerLen - quicAEADTagLen
	lengthField := quicPacketNumberLen + payloadLen + quicAEADTagLen
	payload := make([]byte, 0, payloadLen)
	for _, f := range frags {
		payload = appendCryptoFrame(payload, f)
	}
	require.LessOrEqual(t, len(payload), payloadLen, "frames overflow the Initial")
	for len(payload) < payloadLen {
		payload = append(payload, 0x00)
	}
	header := make([]byte, 0, headerLen)
	header = append(header, 0xC0|byte(quicPacketNumberLen-1))
	header = binary.BigEndian.AppendUint32(header, quicVersion1)
	header = append(header, byte(quicDCIDLen))
	header = append(header, dcid...)
	header = append(header, 0x00, 0x00)
	header = appendQUICVarint(header, uint64(lengthField))
	pnOffset := len(header)
	for i := 0; i < quicPacketNumberLen; i++ {
		header = append(header, byte(pn>>(8*(quicPacketNumberLen-1-i))))
	}
	key, iv, hp := deriveInitialKeys(dcid)
	pkt, err := encryptInitial(header, payload, key, iv, hp, pnOffset, pn)
	require.NoError(t, err)
	return pkt
}

// expBuildInitialTok is expBuildInitial with a non-empty Initial token (as a real
// Chrome carries a ~70-byte Retry/New-Token). tokenLen is a 1-byte varint here
// (fits <64), then token bytes, before the length field.
func expBuildInitialTok(t *testing.T, dcid []byte, pn uint64, ch []byte, off int, token []byte, totalLen int) ([]byte, int) {
	t.Helper()
	require.Less(t, len(token), 64, "token varint stays 1 byte in this helper")
	headerFixed := 1 + 4 + 1 + 1 // first + version + dcidLen + scidLen(0)
	headerLen := headerFixed + quicDCIDLen + 1 + len(token) + 2 + quicPacketNumberLen
	payloadLen := totalLen - headerLen - quicAEADTagLen
	lengthField := quicPacketNumberLen + payloadLen + quicAEADTagLen
	require.True(t, lengthField >= 1<<6 && lengthField < 1<<14)
	rest := ch[off:]
	maxData := payloadLen - (1 + varintLen(uint64(off)) + 2)
	n := len(rest)
	if n > maxData {
		n = maxData
	}
	frag := cryptoFragment{offset: uint64(off), data: rest[:n]}
	payload := appendCryptoFrame(make([]byte, 0, payloadLen), frag)
	for len(payload) < payloadLen {
		payload = append(payload, 0x00)
	}
	header := make([]byte, 0, headerLen)
	header = append(header, 0xC0|byte(quicPacketNumberLen-1))
	header = binary.BigEndian.AppendUint32(header, quicVersion1)
	header = append(header, byte(quicDCIDLen))
	header = append(header, dcid...)
	header = append(header, 0x00) // SCID length 0
	header = append(header, byte(len(token)))
	header = append(header, token...)
	header = appendQUICVarint(header, uint64(lengthField))
	pnOffset := len(header)
	for i := 0; i < quicPacketNumberLen; i++ {
		header = append(header, byte(pn>>(8*(quicPacketNumberLen-1-i))))
	}
	key, iv, hp := deriveInitialKeys(dcid)
	pkt, err := encryptInitial(header, payload, key, iv, hp, pnOffset, pn)
	require.NoError(t, err)
	require.Equal(t, totalLen, len(pkt))
	return pkt, n
}

// TestExpDecodeRealChrome: EXP_PCAP_HEX is the raw bytes (hex) of a real Chrome
// QUIC Initial datagram captured off the wire; decrypt it with Initial keys from
// its DCID and dump the ClientHello extension map, for byte-level comparison with
// our generated decoy. Handles a non-empty token in the header.
func TestExpDecodeRealChrome(t *testing.T) {
	h := strings.TrimSpace(os.Getenv("EXP_PCAP_HEX"))
	if h == "" {
		t.Skip("EXP_PCAP_HEX not set")
	}
	packet, err := hex.DecodeString(h)
	require.NoError(t, err)
	r := bytes.NewReader(packet)
	first, _ := r.ReadByte()
	require.Equal(t, byte(0xC0), first&0xF0, "long header Initial")
	var version uint32
	binary.Read(r, binary.BigEndian, &version)
	dcidLen, _ := r.ReadByte()
	dcid := make([]byte, dcidLen)
	io.ReadFull(r, dcid)
	scidLen, _ := r.ReadByte()
	io.CopyN(io.Discard, r, int64(scidLen))
	tokenLen := readVarint(t, r)
	io.CopyN(io.Discard, r, int64(tokenLen))
	lengthField := readVarint(t, r)
	hdrLen := len(packet) - r.Len()
	t.Logf("version=%08x dcidLen=%d scidLen=%d tokenLen=%d length=%d hdrLen=%d pktLen=%d", version, dcidLen, scidLen, tokenLen, lengthField, hdrLen, len(packet))

	key, iv, hp := deriveInitialKeys(dcid)
	block, err := aes.NewCipher(hp)
	require.NoError(t, err)
	mask := make([]byte, aes.BlockSize)
	block.Encrypt(mask, packet[hdrLen+4:hdrLen+4+aes.BlockSize])
	un := append([]byte(nil), packet...)
	un[0] ^= mask[0] & 0x0f
	pnLen := int(un[0]&0x03) + 1
	var pn uint64
	for i := 0; i < pnLen; i++ {
		un[hdrLen+i] ^= mask[1+i]
		pn = pn<<8 | uint64(un[hdrLen+i])
	}
	aad := un[:hdrLen+pnLen]
	ct := un[hdrLen+pnLen : hdrLen+int(lengthField)]
	cipher := quicAEADAESGCMTLS13(key, iv)
	nonce := make([]byte, cipher.NonceSize())
	binary.BigEndian.PutUint64(nonce[cipher.NonceSize()-8:], pn)
	dec, err := cipher.Open(nil, nonce, ct, aad)
	require.NoError(t, err, "AEAD open — proves it is a real Initial")

	// frame-walk, collect CRYPTO
	var cryptos []cryptoFragment
	fr := bytes.NewReader(dec)
	for {
		ft, e := fr.ReadByte()
		if e == io.EOF {
			break
		}
		switch ft {
		case 0x00, 0x01:
		case 0x06:
			off := readVarint(t, fr)
			ln := readVarint(t, fr)
			idx := len(dec) - fr.Len()
			cryptos = append(cryptos, cryptoFragment{offset: off, data: append([]byte(nil), dec[idx:idx+int(ln)]...)})
			io.CopyN(io.Discard, fr, int64(ln))
		default:
			t.Logf("frame 0x%02x (stop walk)", ft)
			fr = bytes.NewReader(nil)
		}
	}
	// reassemble by offset (sort), contiguous from 0 as far as it goes
	var ch []byte
	for {
		progressed := false
		for _, f := range cryptos {
			if f.offset == uint64(len(ch)) {
				ch = append(ch, f.data...)
				progressed = true
			}
		}
		if !progressed { break }
	}
	t.Logf("CRYPTO frames=%d, CH bytes in this packet=%d", len(cryptos), len(ch))
	if len(ch) < 44 || ch[0] != 0x01 {
		t.Logf("first bytes: % x", ch[:min(32, len(ch))])
		return
	}
	// extension map (only complete within this packet)
	rr := bytes.NewReader(ch[4:])
	skip(t, rr, 2+32)
	sid := readU8(t, rr)
	skip(t, rr, int(sid))
	cs := readU16(t, rr)
	csb := make([]byte, cs)
	io.ReadFull(rr, csb)
	comp := readU8(t, rr)
	skip(t, rr, int(comp))
	extTotal := readU16(t, rr)
	t.Logf("cipher_suites=%dB compression=%dB ext_total=%d", cs, comp, extTotal)
	for rr.Len() > 0 {
		et := readU16(t, rr)
		el := readU16(t, rr)
		if el > uint16(rr.Len()) {
			t.Logf("ext 0x%04x len=%d TRUNCATED at packet boundary", et, el)
			break
		}
		skip(t, rr, int(el))
		t.Logf("ext 0x%04x len=%d", et, el)
	}
}

// TestExpChromeFrameLayout dumps the exact CRYPTO frame layout (offset,len in
// wire order) and token length of each Initial datagram in EXP_PCAP_HEX_LIST
// (newline-separated hex datagrams), to replicate Chrome's real shape.
func TestExpChromeFrameLayout(t *testing.T) {
	list := os.Getenv("EXP_PCAP_HEX_LIST")
	if list == "" {
		t.Skip("EXP_PCAP_HEX_LIST not set")
	}
	for di, h := range strings.Fields(list) {
		packet, err := hex.DecodeString(strings.TrimSpace(h))
		require.NoError(t, err)
		r := bytes.NewReader(packet)
		first, _ := r.ReadByte()
		if first&0xF0 != 0xC0 {
			t.Logf("dg%d: not a long-header Initial (first=0x%02x)", di, first)
			continue
		}
		var version uint32
		binary.Read(r, binary.BigEndian, &version)
		dcidLen, _ := r.ReadByte()
		dcid := make([]byte, dcidLen)
		io.ReadFull(r, dcid)
		scidLen, _ := r.ReadByte()
		io.CopyN(io.Discard, r, int64(scidLen))
		tokenLen := readVarint(t, r)
		io.CopyN(io.Discard, r, int64(tokenLen))
		lengthField := readVarint(t, r)
		hdrLen := len(packet) - r.Len()
		key, iv, hp := deriveInitialKeys(dcid)
		block, _ := aes.NewCipher(hp)
		mask := make([]byte, aes.BlockSize)
		if hdrLen+4+aes.BlockSize > len(packet) {
			t.Logf("dg%d: too short", di)
			continue
		}
		block.Encrypt(mask, packet[hdrLen+4:hdrLen+4+aes.BlockSize])
		un := append([]byte(nil), packet...)
		un[0] ^= mask[0] & 0x0f
		pnLen := int(un[0]&0x03) + 1
		var pn uint64
		for i := 0; i < pnLen; i++ {
			un[hdrLen+i] ^= mask[1+i]
			pn = pn<<8 | uint64(un[hdrLen+i])
		}
		aad := un[:hdrLen+pnLen]
		end := hdrLen + int(lengthField)
		if end > len(un) {
			end = len(un)
		}
		ct := un[hdrLen+pnLen : end]
		cipher := quicAEADAESGCMTLS13(key, iv)
		nonce := make([]byte, cipher.NonceSize())
		binary.BigEndian.PutUint64(nonce[cipher.NonceSize()-8:], pn)
		dec, err := cipher.Open(nil, nonce, ct, aad)
		if err != nil {
			t.Logf("dg%d: AEAD open failed (%v) tokenLen=%d", di, err, tokenLen)
			continue
		}
		var walk []string
		fr := bytes.NewReader(dec)
		pad, ping := 0, 0
		for {
			ft, e := fr.ReadByte()
			if e == io.EOF {
				break
			}
			switch ft {
			case 0x00:
				pad++
			case 0x01:
				ping++
			case 0x06:
				off := readVarint(t, fr)
				ln := readVarint(t, fr)
				io.CopyN(io.Discard, fr, int64(ln))
				walk = append(walk, fmt.Sprintf("CR[off=%d len=%d]", off, ln))
			default:
				walk = append(walk, fmt.Sprintf("?0x%02x", ft))
			}
		}
		t.Logf("dg%d pktLen=%d tokenLen=%d pn=%d frames: %s PAD=%d PING=%d", di, len(packet), tokenLen, pn, strings.Join(walk, " "), pad, ping)
	}
}

// expPadClientHelloTo grows a ClientHello to exactly target bytes by appending a
// TLS padding extension (0x0015) of zeros and fixing the two length fields
// (handshake u24 and extensions u16). Returns the padded ClientHello.
func expPadClientHelloTo(t *testing.T, ch []byte, target int) []byte {
	t.Helper()
	require.GreaterOrEqual(t, target, len(ch)+4, "need room for a padding ext header")
	add := target - len(ch)           // total extra bytes we must add
	padBody := add - 4                // 4 = ext type(2)+len(2)
	require.GreaterOrEqual(t, padBody, 0)
	// extensions_total is a u16 right before the extensions block. Find it:
	// ch[0]=0x01, [1:4]=handshake len, then body. Walk to extensions length.
	r := bytes.NewReader(ch[4:])
	skip(t, r, 2+32)
	sid := readU8(t, r)
	skip(t, r, int(sid))
	cs := readU16(t, r)
	skip(t, r, int(cs))
	comp := readU8(t, r)
	skip(t, r, int(comp))
	extLenPos := 4 + (len(ch) - 4 - r.Len()) // absolute index of the u16 ext-total
	extTotal := int(ch[extLenPos])<<8 | int(ch[extLenPos+1])

	ext := make([]byte, 4+padBody)
	ext[0], ext[1] = 0x00, 0x15 // padding extension
	ext[2], ext[3] = byte(padBody>>8), byte(padBody)
	out := append(append(append([]byte(nil), ch...), ext...))
	// fix extensions_total
	nt := extTotal + 4 + padBody
	out[extLenPos], out[extLenPos+1] = byte(nt>>8), byte(nt)
	// fix handshake length (u24 at [1:4])
	hl := len(out) - 4
	out[1], out[2], out[3] = byte(hl>>16), byte(hl>>8), byte(hl)
	require.Equal(t, target, len(out))
	return out
}

// chromeFrameSpec is the real Chrome 133 first-flight CRYPTO layout captured off
// the wire (dcid a72c…): offsets/lengths per Initial, token 70, with PINGs/PAD.
// ch must be exactly 2501 bytes so these offsets land inside it.
type chromeCR struct{ off, ln, pings int }

var chromePn1 = []chromeCR{{16, 9, 0}, {42, 19, 0}, {0, 3, 0}, {2130, 371, 0}, {14, 2, 0}, {3, 11, 0}, {1979, 151, 0}, {36, 6, 0}, {25, 11, 0}, {1731, 248, 0}}
var chromePn2 = []chromeCR{{61, 1139, 0}}
var chromePn3 = []chromeCR{{1200, 269, 0}, {1469, 147, 0}, {1616, 115, 0}}

func expCRFrame(ch []byte, c chromeCR) []byte {
	return appendCryptoFrame(nil, cryptoFragment{offset: uint64(c.off), data: ch[c.off : c.off+c.ln]})
}

// TestExpArmC133: replicate Chrome's exact three-packet first flight. ch padded
// to 2501, frames placed per the captured map, 70-byte token, PINGs woven in,
// each Initial padded to 1250. EXP_OUT / EXP_DOMAINS. Files C133_<tag>_i1/_i2/_i3.
func TestExpArmC133(t *testing.T) {
	out := os.Getenv("EXP_OUT")
	domains := strings.Fields(os.Getenv("EXP_DOMAINS"))
	if out == "" || len(domains) == 0 {
		t.Skip("EXP_OUT / EXP_DOMAINS not set")
	}
	require.NoError(t, os.MkdirAll(out, 0o755))
	var log strings.Builder
	for _, sni := range domains {
		tag := strings.ReplaceAll(sni, ".", "_")
		raw := expChromePQClientHello(t, sni)
		ch := expPadClientHelloTo(t, raw, 2501)
		require.Equal(t, sni, extractSNI(t, ch), "SNI survives padding")

		dcid := make([]byte, quicDCIDLen)
		_, err := rand.Read(dcid)
		require.NoError(t, err)
		token := make([]byte, 70)
		_, err = rand.Read(token)
		require.NoError(t, err)

		build := func(pn uint64, frames []chromeCR, pings int) []byte {
			var payload []byte
			for _, c := range frames {
				payload = append(payload, expCRFrame(ch, c)...)
			}
			for i := 0; i < pings; i++ {
				payload = append(payload, 0x01)
			}
			return expPackInitialRaw(t, dcid, pn, token, payload, quicInitialTotalLen)
		}
		p1 := build(1, chromePn1, 6)
		p2 := build(2, chromePn2, 0)
		p3 := build(3, chromePn3, 5)
		for i, p := range [][]byte{p1, p2, p3} {
			expWriteCPS(t, out, fmt.Sprintf("C133_%s_i%d", tag, i+1), p)
		}
		fmt.Fprintf(&log, "C133 %s: ch=%d(padded from %d) token=70 dcid=%x pkts=3\n", sni, len(ch), len(raw), dcid)
	}
	require.NoError(t, os.WriteFile(filepath.Join(out, "C133_walk.txt"), []byte(log.String()), 0o644))
	t.Log("\n" + log.String())
}

// expPackInitialRaw packs an arbitrary frame payload into one Initial with a
// token, padding with PADDING(0x00) to totalLen.
func expPackInitialRaw(t *testing.T, dcid []byte, pn uint64, token, frames []byte, totalLen int) []byte {
	t.Helper()
	headerFixed := 1 + 4 + 1 + 1
	tokLenVarint := 1
	if len(token) >= 64 {
		tokLenVarint = 2
	}
	headerLen := headerFixed + quicDCIDLen + tokLenVarint + len(token) + 2 + quicPacketNumberLen
	payloadLen := totalLen - headerLen - quicAEADTagLen
	require.LessOrEqual(t, len(frames), payloadLen, "frames overflow Initial pn=%d", pn)
	payload := append(append([]byte(nil), frames...), make([]byte, payloadLen-len(frames))...)
	lengthField := quicPacketNumberLen + payloadLen + quicAEADTagLen
	header := make([]byte, 0, headerLen)
	header = append(header, 0xC0|byte(quicPacketNumberLen-1))
	header = binary.BigEndian.AppendUint32(header, quicVersion1)
	header = append(header, byte(quicDCIDLen))
	header = append(header, dcid...)
	header = append(header, 0x00)
	header = appendQUICVarint(header, uint64(len(token)))
	header = append(header, token...)
	header = appendQUICVarint(header, uint64(lengthField))
	pnOffset := len(header)
	for i := 0; i < quicPacketNumberLen; i++ {
		header = append(header, byte(pn>>(8*(quicPacketNumberLen-1-i))))
	}
	key, iv, hp := deriveInitialKeys(dcid)
	pkt, err := encryptInitial(header, payload, key, iv, hp, pnOffset, pn)
	require.NoError(t, err)
	require.Equal(t, totalLen, len(pkt))
	return pkt
}
